using Dapper;
using MainPCDoctor.Core.Models;

namespace MainPCDoctor.Storage;

public sealed class SQLiteMetricsStore
{
    private readonly DatabaseFactory _db;

    public SQLiteMetricsStore(DatabaseFactory db)
    {
        _db = db;
    }

    public async Task WriteMinuteAggregateAsync(MinuteAggregate agg, CancellationToken ct = default)
    {
        await using var conn = _db.Open();
        await conn.ExecuteAsync("""
            INSERT INTO metrics_samples_1min
              (sampled_at, cpu_avg_pct, cpu_max_pct,
               mem_available_gb_avg, mem_available_gb_min, mem_commit_ratio_avg, mem_pagefile_pressure,
               disk_util_avg_pct, disk_latency_avg_ms,
               gpu_util_avg_pct, gpu_vram_used_gb_avg, gpu_vram_total_gb)
            VALUES
              (@SampledAt, @CpuAvgPct, @CpuMaxPct,
               @MemAvailGbAvg, @MemAvailGbMin, @MemCommitRatioAvg, @MemPagefilePressure,
               @DiskUtilAvgPct, @DiskLatencyAvgMs,
               @GpuUtilAvgPct, @GpuVramUsedGbAvg, @GpuVramTotalGb)
            """, new
            {
                SampledAt            = agg.SampledAt.ToString("O"),
                agg.CpuAvgPct,
                agg.CpuMaxPct,
                agg.MemAvailGbAvg,
                agg.MemAvailGbMin,
                agg.MemCommitRatioAvg,
                MemPagefilePressure  = agg.MemPagefilePressure ? 1 : 0,
                agg.DiskUtilAvgPct,
                agg.DiskLatencyAvgMs,
                agg.GpuUtilAvgPct,
                agg.GpuVramUsedGbAvg,
                agg.GpuVramTotalGb
            });
    }

    public async Task PurgeOlderThanAsync(int days, CancellationToken ct = default)
    {
        await using var conn = _db.Open();
        await conn.ExecuteAsync(
            "DELETE FROM metrics_samples_1min WHERE sampled_at < datetime('now', @Days)",
            new { Days = $"-{days} days" });
    }
}

public record MinuteAggregate(
    DateTimeOffset SampledAt,
    float  CpuAvgPct,
    float  CpuMaxPct,
    float  MemAvailGbAvg,
    float  MemAvailGbMin,
    float  MemCommitRatioAvg,
    bool   MemPagefilePressure,
    float? DiskUtilAvgPct,
    float? DiskLatencyAvgMs,
    float? GpuUtilAvgPct,
    float? GpuVramUsedGbAvg,
    float? GpuVramTotalGb
);
